// ==UserScript==
// @name         DATN Pixel Drawer - Auto Repair
// @namespace    http://tampermonkey.net/
// @version      13.0
// @description  Auto draw + Tự động sửa pixel bị phá hoại
// @author       You
// @match        https://datn.unifolio.io.vn/*
// @icon         https://www.google.com/s2/favicons?domain=unifolio.io.vn
// @grant        GM_xmlhttpRequest
// @connect      datn.unifolio.io.vn
// ==/UserScript==

(function() {
    'use strict';

    const CONFIG = {
        START_X: 244,
        START_Y: 100,
        TEMPLATE_WIDTH: 79,
        TEMPLATE_HEIGHT: 134,
        MAX_DOTS_PER_WINDOW: 600,
        WINDOW_DURATION: 300000,
        PLACE_PIXEL_URL: 'https://datn.unifolio.io.vn/api/community-pixels',
        GET_PIXEL_URL: 'https://datn.unifolio.io.vn/api/community-pixels',
        LOG_LEVEL: 2,
        CACHE_DURATION: 10000,
        MIN_DELAY: 1000,
        MAX_DELAY: 1800,
        // ===== AUTO REPAIR =====
        AUTO_REPAIR: true,           // Bật/tắt tự động sửa
        REPAIR_INTERVAL: 300000,     // 5 phút quét 1 lần
        REPAIR_BATCH: 10,            // Số pixel sửa mỗi lần
        REPAIR_DELAY: 500,           // Delay giữa các lần sửa
    };

    // ===== LOG =====
    const log = {
        error: (...args) => console.error(...args),
        warn: (...args) => { if (CONFIG.LOG_LEVEL >= 1) console.warn(...args); },
        info: (...args) => { if (CONFIG.LOG_LEVEL >= 2) console.log(...args); },
        debug: (...args) => { if (CONFIG.LOG_LEVEL >= 3) console.log('[DEBUG]', ...args); },
        progress: (placed, total, pct, skipped) => {
            const milestones = [1, 5, 10, 25, 50, 75, 90, 95, 99, 100];
            const current = Math.floor(pct);
            if (milestones.includes(current) && current !== log._lastMilestone) {
                log._lastMilestone = current;
                console.log(`📊 ${placed}/${total} (${pct}%) | Bỏ qua: ${skipped}`);
            }
        },
        _lastMilestone: -1,
        repair: (count, total) => {
            console.log(`🔧 Đã sửa ${count}/${total} pixel bị phá hoại`);
        }
    };

    // ===== CHẶN REQUEST NOTIFICATIONS =====
    const originalFetch = window.fetch;
    window.fetch = function(...args) {
        const url = typeof args[0] === 'string' ? args[0] : args[0]?.url;
        if (url && url.includes('/api/notifications')) {
            return Promise.resolve(new Response(JSON.stringify({ notifications: [] }), {
                status: 200,
                headers: { 'Content-Type': 'application/json' }
            }));
        }
        return originalFetch.apply(this, args);
    };

    let isRunning = false;
    let isPaused = false;
    let pixelQueue = [];
    let totalPixels = 0;
    let dotsPlacedInWindow = 0;
    let windowStartTime = Date.now();
    let stats = {
        totalPlaced: 0,
        totalSkipped: 0,
        totalRepaired: 0,
        failedAttempts: 0,
        windowsCompleted: 0,
        startTime: null,
        longBreaks: 0,
    };

    let existingPixels = new Map();
    let lastFetchTime = 0;
    let templatePixels = []; // Lưu template để so sánh
    let repairTimer = null;
    let isRepairing = false;

    // ===== LẤY PIXEL =====
    async function fetchExistingPixels(force = false) {
        const now = Date.now();
        if (!force && now - lastFetchTime < CONFIG.CACHE_DURATION && existingPixels.size > 0) {
            return existingPixels;
        }

        try {
            const response = await fetch(CONFIG.GET_PIXEL_URL + '?since=0', {
                method: 'GET',
                headers: {
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                },
                credentials: 'include'
            });

            if (!response.ok) {
                log.warn('⚠️ Không lấy được pixel');
                return existingPixels;
            }

            const data = await response.json();
            existingPixels = new Map();

            let pixels = [];
            if (data.pixels && Array.isArray(data.pixels)) {
                pixels = data.pixels;
            } else if (Array.isArray(data)) {
                pixels = data;
            }

            pixels.forEach(p => {
                let x, y, color;
                if (Array.isArray(p) && p.length >= 3) {
                    x = p[0];
                    y = p[1];
                    color = p[2];
                } else if (typeof p === 'object' && p !== null) {
                    x = p.x;
                    y = p.y;
                    color = p.color || p.color_id || '#FFFFFF';
                }

                if (x !== undefined && y !== undefined && color) {
                    const key = `${x},${y}`;
                    existingPixels.set(key, color);
                }
            });

            lastFetchTime = now;
            return existingPixels;

        } catch (error) {
            log.error('❌ Lỗi lấy pixel:', error);
            return existingPixels;
        }
    }

    function isPixelExists(x, y, color) {
        const key = `${x},${y}`;
        const existing = existingPixels.get(key);
        if (!existing) return false;
        return existing.toUpperCase() === color.toUpperCase();
    }

    // ===== AUTO REPAIR - QUÉT VÀ SỬA PIXEL BỊ PHÁ =====
    async function scanAndRepair() {
        if (isRepairing || !CONFIG.AUTO_REPAIR || templatePixels.length === 0) {
            return;
        }

        isRepairing = true;
        log.info('🔍 Đang quét pixel bị phá hoại...');

        try {
            // Lấy pixel hiện tại từ server
            await fetchExistingPixels(true);

            let damagedPixels = [];

            // So sánh từng pixel trong template
            for (const p of templatePixels) {
                const globalX = CONFIG.START_X + p.x;
                const globalY = CONFIG.START_Y + p.y;
                const key = `${globalX},${globalY}`;
                const currentColor = existingPixels.get(key);

                // Nếu pixel chưa có hoặc khác màu → bị phá
                if (!currentColor || currentColor.toUpperCase() !== p.color.toUpperCase()) {
                    damagedPixels.push(p);
                }
            }

            if (damagedPixels.length === 0) {
                log.info('✅ Không có pixel nào bị phá hoại!');
                isRepairing = false;
                return;
            }

            log.warn(`⚠️ Phát hiện ${damagedPixels.length} pixel bị phá hoại! Đang sửa...`);

            // Sửa từng pixel bị phá
            let repaired = 0;
            const batchSize = CONFIG.REPAIR_BATCH;

            for (let i = 0; i < damagedPixels.length; i += batchSize) {
                const batch = damagedPixels.slice(i, i + batchSize);

                for (const p of batch) {
                    const globalX = CONFIG.START_X + p.x;
                    const globalY = CONFIG.START_Y + p.y;

                    // Kiểm tra lại trước khi sửa (phòng trường hợp đã được sửa)
                    const key = `${globalX},${globalY}`;
                    const currentColor = existingPixels.get(key);
                    if (currentColor && currentColor.toUpperCase() === p.color.toUpperCase()) {
                        continue;
                    }

                    const result = await placePixelWithRetry(globalX, globalY, p.color);
                    if (result.success) {
                        repaired++;
                        existingPixels.set(key, p.color);
                        log.debug(`🔧 Sửa pixel (${globalX},${globalY})`);
                    }

                    // Delay giữa các lần sửa
                    await new Promise(resolve => setTimeout(resolve, CONFIG.REPAIR_DELAY));
                }

                // Cập nhật progress
                log.repair(repaired, damagedPixels.length);

                // Refresh cache sau mỗi batch
                if (i + batchSize < damagedPixels.length) {
                    await fetchExistingPixels(true);
                }
            }

            stats.totalRepaired += repaired;
            log.info(`✅ Đã sửa ${repaired}/${damagedPixels.length} pixel bị phá hoại!`);

            // Cập nhật UI
            updateStats();

        } catch (error) {
            log.error('❌ Lỗi khi quét/sửa pixel:', error);
        }

        isRepairing = false;
    }

    // ===== ĐẶT PIXEL CÓ RETRY =====
    async function placePixelWithRetry(x, y, color, maxRetries = 3) {
        for (let attempt = 1; attempt <= maxRetries; attempt++) {
            try {
                const payload = { changes: [{ x, y, color }] };

                const response = await fetch(CONFIG.PLACE_PIXEL_URL, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'Accept': 'application/json',
                        'X-Requested-With': 'XMLHttpRequest',
                    },
                    body: JSON.stringify(payload),
                    credentials: 'include'
                });

                if (response.ok) {
                    return { success: true };
                }

                if (response.status === 429) {
                    log.warn(`⏳ Quota hết, đợi 5 phút...`);
                    await new Promise(resolve => setTimeout(resolve, CONFIG.WINDOW_DURATION));
                    continue;
                }

                const errorData = await response.json();
                log.debug(`❌ Lỗi (lần ${attempt}):`, errorData);

            } catch (error) {
                log.debug(`❌ Lỗi (lần ${attempt}):`, error.message);
            }

            // Đợi trước khi thử lại
            await new Promise(resolve => setTimeout(resolve, 2000));
        }

        return { success: false };
    }

    // ===== BẮT ĐẦU AUTO REPAIR =====
    function startAutoRepair() {
        if (repairTimer) {
            clearInterval(repairTimer);
        }

        if (CONFIG.AUTO_REPAIR && templatePixels.length > 0) {
            log.info(`🛡️ Auto Repair: Mỗi ${CONFIG.REPAIR_INTERVAL/60000} phút`);
            repairTimer = setInterval(() => {
                if (isRunning && !isPaused) {
                    scanAndRepair();
                }
            }, CONFIG.REPAIR_INTERVAL);

            // Quét lần đầu sau 30 giây
            setTimeout(() => {
                if (isRunning) {
                    scanAndRepair();
                }
            }, 30000);
        }
    }

    // ===== DELAY =====
    function getHumanLikeDelay() {
        const progress = totalPixels > 0 ? stats.totalPlaced / totalPixels : 0;
        let baseDelay;
        if (progress < 0.1) baseDelay = 400 + Math.random() * 400;
        else if (progress < 0.3) baseDelay = 500 + Math.random() * 500;
        else if (progress < 0.6) baseDelay = 600 + Math.random() * 600;
        else if (progress < 0.85) baseDelay = 800 + Math.random() * 800;
        else baseDelay = 500 + Math.random() * 500;
        return Math.min(Math.max(baseDelay, CONFIG.MIN_DELAY), CONFIG.MAX_DELAY);
    }

    function shouldTakeLongBreak() {
        const interval = 50 + Math.floor(Math.random() * 30);
        if (stats.totalPlaced > 0 && stats.totalPlaced % interval === 0) {
            return 30000 + Math.random() * 60000;
        }
        return 0;
    }

    // ===== LOAD ẢNH =====
    function loadTemplateFromImage(imageUrl) {
        return new Promise((resolve, reject) => {
            log.info('🔄 Đang load ảnh...');
            const img = new Image();
            img.crossOrigin = 'Anonymous';

            const timeout = setTimeout(() => {
                reject(new Error('❌ Load ảnh timeout (15s)'));
            }, 15000);

            img.onload = function() {
                clearTimeout(timeout);
                const TARGET_WIDTH = CONFIG.TEMPLATE_WIDTH;
                const TARGET_HEIGHT = CONFIG.TEMPLATE_HEIGHT;

                const canvas = document.createElement('canvas');
                canvas.width = TARGET_WIDTH;
                canvas.height = TARGET_HEIGHT;
                const ctx = canvas.getContext('2d');
                ctx.imageSmoothingEnabled = false;
                ctx.drawImage(img, 0, 0, TARGET_WIDTH, TARGET_HEIGHT);

                const imageData = ctx.getImageData(0, 0, TARGET_WIDTH, TARGET_HEIGHT).data;
                const pixels = [];

                for (let y = 0; y < TARGET_HEIGHT; y++) {
                    for (let x = 0; x < TARGET_WIDTH; x++) {
                        const index = (y * TARGET_WIDTH + x) * 4;
                        const r = imageData[index];
                        const g = imageData[index + 1];
                        const b = imageData[index + 2];
                        const a = imageData[index + 3];
                        if (a < 128) continue;
                        const hex = '#' +
                            r.toString(16).padStart(2, '0') +
                            g.toString(16).padStart(2, '0') +
                            b.toString(16).padStart(2, '0');
                        pixels.push({ x, y, color: hex.toUpperCase() });
                    }
                }

                log.info(`✅ Loaded ${pixels.length} pixels`);
                templatePixels = pixels; // Lưu template cho auto repair
                resolve(pixels);
            };

            img.onerror = function() {
                clearTimeout(timeout);
                log.error('❌ Load ảnh thất bại!');
                reject(new Error('❌ Không thể load ảnh!'));
            };
            img.src = imageUrl;
        });
    }

    // ===== API =====
    async function placePixels(pixels) {
        const p = pixels[0];
        const globalX = CONFIG.START_X + p.x;
        const globalY = CONFIG.START_Y + p.y;
        const color = p.color;

        if (isPixelExists(globalX, globalY, color)) {
            return { success: true, skipped: true, count: 1 };
        }

        try {
            const payload = { changes: [{ x: globalX, y: globalY, color }] };

            const response = await fetch(CONFIG.PLACE_PIXEL_URL, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Accept': 'application/json',
                    'X-Requested-With': 'XMLHttpRequest',
                },
                body: JSON.stringify(payload),
                credentials: 'include'
            });

            if (!response.ok) {
                const errorData = await response.json();
                if (response.status === 429) {
                    return { success: false, rateLimited: true, error: errorData.error };
                }
                throw new Error(errorData.error || `HTTP ${response.status}`);
            }

            existingPixels.set(`${globalX},${globalY}`, color);
            return { success: true, skipped: false, count: 1 };

        } catch (error) {
            log.error('❌ Lỗi API:', error.message);
            return { success: false, rateLimited: false, error: error.message };
        }
    }

    // ===== VẼ =====
    async function steadyMode() {
        let pixelCount = 0;
        let lastCacheRefresh = Date.now();

        while (isRunning && !isPaused && pixelQueue.length > 0) {
            const now = Date.now();

            if (now - lastCacheRefresh > 30000) {
                await fetchExistingPixels(true);
                lastCacheRefresh = now;
            }

            if (now - windowStartTime >= CONFIG.WINDOW_DURATION) {
                dotsPlacedInWindow = 0;
                windowStartTime = now;
                stats.windowsCompleted++;
                updateStats();
            }

            if (dotsPlacedInWindow >= CONFIG.MAX_DOTS_PER_WINDOW) {
                const waitTime = CONFIG.WINDOW_DURATION - (now - windowStartTime);
                const waitSeconds = Math.ceil(waitTime / 1000);
                log.info(`⏳ Đợi ${waitSeconds}s`);
                updateStatus(`⏳ Đợi ${waitSeconds}s | Còn: ${pixelQueue.length}`);
                await new Promise(resolve => setTimeout(resolve, waitTime + 1000));
                continue;
            }

            const pixel = pixelQueue.shift();
            if (!pixel) break;
            pixelCount++;

            const result = await placePixels([pixel]);

            if (result.success) {
                if (result.skipped) {
                    stats.totalSkipped++;
                } else {
                    dotsPlacedInWindow++;
                    stats.totalPlaced++;
                }
                updateStats();

                const progress = totalPixels > 0 ? ((stats.totalPlaced / totalPixels) * 100) : 0;

                if (pixelCount % 30 === 0 || pixelCount <= 10) {
                    log.progress(stats.totalPlaced, totalPixels, progress, stats.totalSkipped);
                }

                if (!result.skipped) {
                    const delay = getHumanLikeDelay();
                    await new Promise(resolve => setTimeout(resolve, delay));
                } else {
                    await new Promise(resolve => setTimeout(resolve, 50 + Math.random() * 100));
                }

                const breakTime = shouldTakeLongBreak();
                if (breakTime > 0) {
                    stats.longBreaks++;
                    log.info(`☕ Nghỉ ${Math.round(breakTime/1000)}s`);
                    updateStatus(`☕ Nghỉ ${Math.round(breakTime/1000)}s`);
                    await new Promise(resolve => setTimeout(resolve, breakTime));
                }

            } else if (result.rateLimited) {
                log.warn('🚫 Hết quota');
                pixelQueue.unshift(pixel);
                dotsPlacedInWindow = CONFIG.MAX_DOTS_PER_WINDOW;
            } else {
                log.error('❌ Lỗi, thử lại');
                pixelQueue.unshift(pixel);
                stats.failedAttempts++;
                updateStats();
                await new Promise(resolve => setTimeout(resolve, 5000));
            }
        }

        if (pixelQueue.length === 0) {
            log.info(`🎉 HOÀN THÀNH! Đã vẽ: ${stats.totalPlaced}, Bỏ qua: ${stats.totalSkipped}`);
            updateStatus(`🎉 HOÀN THÀNH!`);
            stopDrawing();
        }
    }

    // ===== ĐIỀU KHIỂN =====
    async function startDrawing(templateUrl) {
        if (isRunning) return;

        log.info('🎨 Bắt đầu...');
        updateStatus('🔄 Load template...');

        try {
            const pixels = await loadTemplateFromImage(templateUrl);

            await fetchExistingPixels(true);

            let skippedCount = 0;
            const filteredPixels = [];

            for (const p of pixels) {
                const globalX = CONFIG.START_X + p.x;
                const globalY = CONFIG.START_Y + p.y;
                if (isPixelExists(globalX, globalY, p.color)) {
                    skippedCount++;
                } else {
                    filteredPixels.push(p);
                }
            }

            log.info(`📊 Tổng: ${pixels.length} | Đã có: ${skippedCount} | Cần vẽ: ${filteredPixels.length}`);

            if (filteredPixels.length === 0) {
                log.info('🎉 Tất cả pixel đã được vẽ xong!');
                updateStatus('🎉 HOÀN THÀNH!');
                return;
            }

            pixelQueue = [...filteredPixels];
            totalPixels = pixels.length;
            stats = {
                totalPlaced: 0,
                totalSkipped: skippedCount,
                totalRepaired: 0,
                failedAttempts: 0,
                windowsCompleted: 0,
                startTime: Date.now(),
                longBreaks: 0,
            };
            dotsPlacedInWindow = 0;
            windowStartTime = Date.now();
            isRunning = true;
            isPaused = false;

            createControlPanel();
            updateStats();

            // Khởi động Auto Repair
            startAutoRepair();

            steadyMode();

        } catch (error) {
            log.error('❌ Lỗi:', error.message);
            alert('❌ ' + error.message);
        }
    }

    // ===== UI =====

        removeControlPanel();
        const panel = document.createElement('div');
        panel.id = 'pixel-drawer-panel';
        panel.style.cssText = `
            position: fixed; top: 20px; right: 20px;
            background: linear-gradient(135deg, #1a1a2e 0%, #16213e 100%);
            color: white; padding: 15px; border-radius: 12px;
            z-index: 999999; font-family: Arial, sans-serif;
            font-size: 13px; min-width: 220px;
            box-shadow: 0 10px 25px rgba(0,0,0,0.5);
            border: 1px solid rgba(255,255,255,0.1);
        `;
        panel.innerHTML = `
            <div style="font-size:16px;font-weight:bold;margin-bottom:10px;color:#4CAF50;text-align:center;">🎨 DATN</div>
            <div id="drawer-status" style="background:rgba(0,0,0,0.3);padding:8px;border-radius:6px;margin-bottom:10px;text-align:center;color:#FFC107;">🚀 Đang chạy...</div>
            <div style="background:rgba(0,0,0,0.2);padding:10px;border-radius:6px;margin-bottom:10px;">
                <div style="display:flex;justify-content:space-between;"><span>📊</span><span id="progress-text" style="color:#4CAF50;">0%</span></div>
                <div style="background:rgba(255,255,255,0.1);border-radius:8px;height:4px;margin-top:4px;"><div id="progress-bar" style="background:linear-gradient(90deg,#4CAF50,#8BC34A);height:100%;width:0%;border-radius:8px;"></div></div>
                <div style="display:grid;grid-template-columns:1fr 1fr 1fr;gap:4px;margin-top:8px;font-size:11px;">
                    <div>✅ <span id="placed-count" style="color:#4CAF50;">0</span></div>
                    <div>⏭️ <span id="skipped-count" style="color:#FFC107;">0</span></div>
                    <div>📦 <span id="remaining-count" style="color:#2196F3;">0</span></div>
                    <div>🔧 <span id="repaired-count" style="color:#FF6B6B;">0</span></div>
                    <div>🎯 <span id="quota-info" style="color:#FF9800;">0/600</span></div>
                    <div>❌ <span id="error-count" style="color:#F44336;">0</span></div>
                </div>
            </div>
            <div style="display:flex;gap:6px;">
                <button id="pause-resume-btn" style="flex:1;padding:8px;background:#FF9800;color:white;border:none;border-radius:6px;cursor:pointer;font-weight:bold;">⏸️</button>
                <button id="stop-drawing-btn" style="flex:1;padding:8px;background:#f44336;color:white;border:none;border-radius:6px;cursor:pointer;font-weight:bold;">⏹</button>
            </div>
            <div style="margin-top:8px;font-size:10px;color:#666;text-align:center;">Space: Pause | Esc: Stop | 🔧 Auto Repair: ${CONFIG.AUTO_REPAIR ? 'Bật' : 'Tắt'}</div>
        `;
        document.body.appendChild(panel);

        document.getElementById('pause-resume-btn').addEventListener('click', function() {
            if (isPaused) { resumeDrawing(); this.textContent = '⏸️'; this.style.background = '#FF9800'; }
            else { pauseDrawing(); this.textContent = '▶️'; this.style.background = '#2196F3'; }
        });
        document.getElementById('stop-drawing-btn').addEventListener('click', stopDrawing);
    }

    function updateStats() {
        const progress = totalPixels > 0 ? ((stats.totalPlaced / totalPixels) * 100).toFixed(1) : 0;
        document.getElementById('progress-text').textContent = `${progress}%`;
        document.getElementById('progress-bar').style.width = `${progress}%`;
        document.getElementById('placed-count').textContent = stats.totalPlaced;
        document.getElementById('skipped-count').textContent = stats.totalSkipped;
        document.getElementById('repaired-count').textContent = stats.totalRepaired;
        document.getElementById('remaining-count').textContent = pixelQueue.length;
        document.getElementById('quota-info').textContent = `${dotsPlacedInWindow}/${CONFIG.MAX_DOTS_PER_WINDOW}`;
        document.getElementById('error-count').textContent = stats.failedAttempts;
    }

    function updateStatus(message) {
        const el = document.getElementById('drawer-status');
        if (el) el.textContent = message;
    }

    function removeControlPanel() {
        const panel = document.getElementById('pixel-drawer-panel');
        if (panel) panel.remove();
    }

    function pauseDrawing() {
        isPaused = true;
        updateStatus('⏸️ TẠM DỪNG');
    }

    function resumeDrawing() {
        isPaused = false;
        updateStatus('▶️ Đang vẽ...');
        steadyMode();
    }

    function stopDrawing() {
        isRunning = false;
        isPaused = false;
        if (repairTimer) {
            clearInterval(repairTimer);
            repairTimer = null;
        }
        log.info(`🛑 Dừng: ${stats.totalPlaced}/${totalPixels} | Đã sửa: ${stats.totalRepaired}`);
        updateStatus(`🛑 ${stats.totalPlaced}/${totalPixels}`);
        setTimeout(removeControlPanel, 10000);
    }

    // ===== PHÍM TẮT =====
    document.addEventListener('keydown', function(e) {
        if (e.ctrlKey && e.shiftKey && e.key === 'D') {
            e.preventDefault();
            if (isRunning) { stopDrawing(); }
            else {
                const url = prompt('🔗 URL ảnh (hoặc paste base64):', '');
                if (url) startDrawing(url);
            }
        }
        if (e.key === 'Escape' && isRunning) { e.preventDefault(); stopDrawing(); }
        if (e.key === ' ' && isRunning) {
            e.preventDefault();
            if (isPaused) resumeDrawing();
            else pauseDrawing();
        }
        // Phím R: Quét sửa ngay lập tức
        if (e.key === 'r' && e.ctrlKey && isRunning) {
            e.preventDefault();
            log.info('🔧 Quét sửa thủ công...');
            scanAndRepair();
        }
    });

    console.log('🎨 DATN Pixel Drawer - Auto Repair');
    console.log('========================================');
    console.log('🛡️ Tự động phát hiện và sửa pixel bị phá hoại');
    console.log(`⏱️ Quét mỗi ${CONFIG.REPAIR_INTERVAL/60000} phút`);
    console.log(`🔧 Sửa tối đa ${CONFIG.REPAIR_BATCH} pixel/lần`);
    console.log('📋 Phím Ctrl+R: Quét sửa thủ công');
    console.log('========================================');
    console.log('⌨️ Ctrl+Shift+D: Bắt đầu');
    console.log('========================================');

})();
